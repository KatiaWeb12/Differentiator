\begin{task}{1}
\[
\frac{\sin({x}^{2})}{\ln(x + 1)}\]
\end{task}

\begin{solution}
\[
(x + 1)' = 1 + 0
\]

\[
(\ln(x + 1))' = \frac{1 + 0}{x + 1}
\]

\[
({x}^{2})' = 2 \cdot {x}^{2 - 1} \cdot 1
\]

\[
(\sin({x}^{2}))' = \cos({x}^{2}) \cdot 2 \cdot {x}^{2 - 1} \cdot 1
\]

\[
(\frac{\sin({x}^{2})}{\ln(x + 1)})' = \frac{\cos({x}^{2}) \cdot 2 \cdot {x}^{2 - 1} \cdot 1 \cdot \ln(x + 1) - \sin({x}^{2}) \cdot (\frac{1 + 0}{x + 1})}{\ln(x + 1) \cdot \ln(x + 1)}
\]

\[
2 - 1 \longrightarrow 1
\]

\[
{x}^{1} \longrightarrow x
\]

\[
2 \cdot x \cdot 1 \longrightarrow 2 \cdot x
\]

\[
1 + 0 \longrightarrow 1
\]

\[
\ln(x + 1) \cdot \ln(x + 1) \longrightarrow {(\ln(x + 1))}^{2}
\]

Итоговый ответ: \\
\[
\boxed{
\frac{\cos({x}^{2}) \cdot 2 \cdot x \cdot \ln(x + 1) - \sin({x}^{2}) \cdot (\frac{1}{x + 1})}{{(\ln(x + 1))}^{2}}
}
\]
\end{solution}
